Exemplo: adicionar módulo de Produtos numa pet shop
São 4 passos sempre na mesma ordem:

PASSO 1 — Cria o Model models/Produto.js
jsconst mongoose = require('mongoose');

const produtoSchema = new mongoose.Schema({
  nome: { type: String, required: true },
  tipo: { type: String, required: true },   // ex: ração, brinquedo, remédio
  preco: { type: Number, required: true },
  qtde_disponivel: { type: Number, required: true, default: 0 }
});

module.exports = mongoose.model('Produto', produtoSchema);

PASSO 2 — Cria o Routes routes/produtos.js
jsconst express = require('express');
const router = express.Router();
const Produto = require('../models/Produto');

// Middleware de login
function verificarLogin(req, res, next) {
  if (req.session.usuario) {
    next();
  } else {
    res.redirect('/usuarios/login');
  }
}

// READ — Listagem pública
router.get('/', async (req, res) => {
  const produtos = await Produto.find();
  res.render('produtos-lista', { produtos });
});

// READ — Página de gerência
router.get('/gerencia', verificarLogin, async (req, res) => {
  const produtos = await Produto.find();
  res.render('produtos-gerencia', { produtos, mensagem: null });
});

// CREATE — Cadastrar
router.post('/cadastrar', verificarLogin, async (req, res) => {
  const { nome, tipo, preco, qtde_disponivel } = req.body;
  const novoProduto = new Produto({ nome, tipo, preco, qtde_disponivel });
  await novoProduto.save();
  res.redirect('/produtos/gerencia');
});

// DELETE — Remover
router.post('/remover/:id', verificarLogin, async (req, res) => {
  await Produto.findByIdAndDelete(req.params.id);
  res.redirect('/produtos/gerencia');
});

// READ — Página de editar
router.get('/editar/:id', verificarLogin, async (req, res) => {
  const produto = await Produto.findById(req.params.id);
  res.render('produto-editar', { produto, mensagem: null });
});

// UPDATE — Atualizar
router.post('/atualizar/:id', verificarLogin, async (req, res) => {
  const { nome, tipo, preco, qtde_disponivel } = req.body;
  await Produto.findByIdAndUpdate(req.params.id, { nome, tipo, preco, qtde_disponivel });
  res.redirect('/produtos/gerencia');
});

// UPDATE — Vender/decrementar
router.post('/vender/:id', verificarLogin, async (req, res) => {
  const produto = await Produto.findById(req.params.id);
  if (produto && produto.qtde_disponivel > 0) {
    produto.qtde_disponivel -= 1;
    await produto.save();
  }
  res.redirect('/produtos/gerencia');
});

module.exports = router;

PASSO 3 — Registra no server.js
Só adiciona 2 linhas no server.js:
js// ANTES — só tinha animais
const animaisRouter = require('./routes/animais');
app.use('/animais', animaisRouter);

// DEPOIS — adiciona produtos também
const animaisRouter = require('./routes/animais');
const produtosRouter = require('./routes/produtos');  // ← linha nova

app.use('/animais', animaisRouter);
app.use('/produtos', produtosRouter);  // ← linha nova

Resumo dos 4 passos sempre na mesma ordem
1. models/     → cria o arquivo com os campos
2. routes/     → cria o arquivo com as ações (CRUD)
3. server.js   → registra a rota nova (2 linhas)
4. views/      → cria os 3 arquivos .ejs (lista, gerencia, editar)
Seguindo essa ordem você consegue adicionar quantos módulos quiser sem perder nada!
