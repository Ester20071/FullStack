  // Acha o carro pelo ID e atualiza os campos

VENDER — Decrementa quantidade
jsrouter.post('/vender/:id', verificarLogin, async (req, res) => {
  const carro = await Carro.findById(req.params.id); // acha o carro
  if (carro && carro.qtde_disponivel > 0) {           // se tiver estoque
    carro.qtde_disponivel -= 1;                       // tira 1
    await carro.save();                               // salva
  }
  res.redirect('/carros/gerencia');
